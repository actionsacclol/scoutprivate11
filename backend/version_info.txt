# PyInstaller Windows version resource for krypt-scout-backend.exe.
#
# An unsigned exe with NO version/company metadata is a textbook AV heuristic
# trigger — "unknown binary, no provenance". Embedding real VERSIONINFO makes
# the backend look like the ordinary application it is and measurably lowers
# false-positive quarantine risk (which, with no code-signing cert, is the one
# AV lever we still have). Wired into the .spec via EXE(version=...).
#
# Keep filevers/prodvers in sync with package.json "version" (0.3.2 -> 0,3,2,0).
# test_version_sync.py fails the build if they drift — which matters because
# package.json's version is what CI turns into the release tag.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 3, 2, 0),
    prodvers=(0, 3, 2, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,      # VOS_NT_WINDOWS32
    fileType=0x1,    # VFT_APP
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',  # US English, Unicode
        [
          StringStruct('CompanyName', 'Krypt'),
          StringStruct('FileDescription', 'Krypt Scout backend'),
          StringStruct('FileVersion', '0.3.2.0'),
          StringStruct('InternalName', 'krypt-scout-backend'),
          StringStruct('LegalCopyright', 'Copyright (c) Krypt'),
          StringStruct('OriginalFilename', 'krypt-scout-backend.exe'),
          StringStruct('ProductName', 'Krypt Scout'),
          StringStruct('ProductVersion', '0.3.2.0'),
        ],
      ),
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])]),
  ],
)
